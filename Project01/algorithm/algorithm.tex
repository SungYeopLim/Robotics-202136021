\begin{algorithm}
\caption{LiDAR Obstacle Avoidance with Stop and Turn Lock}
\begin{algorithmic}[1]
    \Function{SectorMin}{$a, b$} \Comment{ignore invalid distance 0}
        \State \Return $\min\{\, D[\theta] \mid a \le \theta \le b,\ D[\theta] > 0 \,\}$
    \EndFunction
    \Statex
    \Procedure{ObstacleAvoidance}{}
        \State $state \gets FORWARD$, \; $dir \gets NONE$
        \While{$true$}
            \State $d_F \gets \Call{SectorMin}{150, 210}$ \Comment{read LiDAR every cycle}
            \State $d_L \gets \Call{SectorMin}{60, 120}$
            \State $d_R \gets \Call{SectorMin}{240, 300}$
            \If{$state = TURN$} \Comment{keep the chosen direction}
                \If{$d_F > CLEAR\_DISTANCE$}
                    \State $state \gets FORWARD$
                \EndIf
            \ElsIf{$d_F < STOP\_DISTANCE$} \Comment{stop first, then decide}
                \State $\textproc{SetVelocity}(0, 0)$, \; wait $T\_STOP$
                \If{$d_L > d_R$}
                    \State $dir \gets LEFT$
                \Else
                    \State $dir \gets RIGHT$
                \EndIf
                \State $state \gets TURN$
            \EndIf
            \If{$state = FORWARD$}
                \State $v_L \gets BASE\_SPEED$, \; $v_R \gets BASE\_SPEED$
            \ElsIf{$dir = LEFT$}
                \State $v_L \gets -TURN\_SPEED$, \; $v_R \gets TURN\_SPEED$
            \Else
                \State $v_L \gets TURN\_SPEED$, \; $v_R \gets -TURN\_SPEED$
            \EndIf
            \State $\textproc{SetVelocity}(v_L, v_R)$
        \EndWhile
    \EndProcedure
\end{algorithmic}
\end{algorithm}
